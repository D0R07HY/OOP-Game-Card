import java.awt.*;
import java.awt.event.MouseAdapter;
import java.awt.event.MouseEvent;
import java.util.*;
import java.util.List;
import java.util.function.Consumer;
import javax.swing.*;

public class BoardPanel extends JPanel {

    private List<Player> players = new ArrayList<>();
    private Map<Player, Integer> deckCounts = new HashMap<>();
    private final Map<StarCard, Rectangle> starBounds = new HashMap<>();
    private final Map<Player, Rectangle> playerBounds = new HashMap<>();
    private final Set<StarCard> selectableStars = new HashSet<>();
    private final Set<Player> selectablePlayers = new HashSet<>();
    private Consumer<StarCard> onStarClicked;
    private Consumer<Player> onPlayerClicked;
    private Player localPlayer;

    public BoardPanel() {
        setBackground(Color.LIGHT_GRAY);
        installMouseHandling();
    }

    public void setLocalPlayer(Player player) {
        this.localPlayer = player;
        repaint();
    }

    private void installMouseHandling() {
        MouseAdapter mouse = new MouseAdapter() {
            @Override
            public void mouseClicked(MouseEvent e) {
                for (Map.Entry<StarCard, Rectangle> entry : starBounds.entrySet()) {
                    if (entry.getValue().contains(e.getPoint()) && selectableStars.contains(entry.getKey())) {
                        if (onStarClicked != null) onStarClicked.accept(entry.getKey());
                        return;
                    }
                }
                for (Map.Entry<Player, Rectangle> entry : playerBounds.entrySet()) {
                    if (entry.getValue().contains(e.getPoint()) && selectablePlayers.contains(entry.getKey())) {
                        if (onPlayerClicked != null) onPlayerClicked.accept(entry.getKey());
                        return;
                    }
                }
            }

            @Override
            public void mouseMoved(MouseEvent e) {
                boolean target = starBounds.entrySet().stream().anyMatch(entry -> entry.getValue().contains(e.getPoint()) && selectableStars.contains(entry.getKey()))
                        || playerBounds.entrySet().stream().anyMatch(entry -> entry.getValue().contains(e.getPoint()) && selectablePlayers.contains(entry.getKey()));
                setCursor(Cursor.getPredefinedCursor(target ? Cursor.HAND_CURSOR : Cursor.DEFAULT_CURSOR));
            }
        };
        addMouseListener(mouse);
        addMouseMotionListener(mouse);
    }

    public void setSelectable(Set<StarCard> stars, Set<Player> players) {
        selectableStars.clear();
        selectablePlayers.clear();
        if (stars != null) selectableStars.addAll(stars);
        if (players != null) selectablePlayers.addAll(players);
        repaint();
    }

    public void clearSelectable() {
        setSelectable(null, null);
    }

    public void setOnStarClicked(Consumer<StarCard> l) { this.onStarClicked = l; }
    public void setOnPlayerClicked(Consumer<Player> l) { this.onPlayerClicked = l; }

    public void playCardToDropZone(Card card, Image cardImg) { repaint(); }
    public void setPlayers(List<Player> players) { this.players = (players == null) ? new ArrayList<>() : players; repaint(); }
    public void setDeckCounts(Map<Player, Integer> deckCounts) { this.deckCounts = (deckCounts == null) ? new HashMap<>() : deckCounts; repaint(); }

    @Override
    protected void paintComponent(Graphics g) {
        super.paintComponent(g);
        starBounds.clear();
        playerBounds.clear();

        int w = getWidth();
        int h = getHeight();


        if (players == null || players.isEmpty()) return;

        
        List<Player> orderedPlayers = new ArrayList<>();
        if (localPlayer != null && players.contains(localPlayer)) {
            orderedPlayers.add(localPlayer);
            for (Player p : players) {
                if (p != localPlayer) {
                    orderedPlayers.add(p);
                }
            }
        } else {
            orderedPlayers.addAll(players);
        }

        int totalPlayers = Math.min(orderedPlayers.size(), 4);

        
        for (int i = 0; i < totalPlayers; i++) {
            Player p = orderedPlayers.get(i);
            if (p == null) continue;

            int px = 20, py = 20;
            if (i == 0) { px = w / 2 - 225; py = h - 100; }     
            else if (i == 1) { px = w / 2 - 225; py = 20; }     
            else if (i == 2) { px = 20; py = h / 2 - 50; }     
            else if (i == 3) { px = w - 160; py = h / 2 - 50; } 

            
            Rectangle pRect = new Rectangle(px, py, 140, 60);
            playerBounds.put(p, pRect);
            g.setColor(selectablePlayers.contains(p) ? Color.YELLOW : Color.WHITE);
            g.fillRect(pRect.x, pRect.y, pRect.width, pRect.height);
            g.setColor(Color.BLACK);
            g.drawRect(pRect.x, pRect.y, pRect.width, pRect.height);

            Integer deck = deckCounts.get(p);
            g.drawString(p.getUsername() + (!p.isActive() ? " (OUT)" : ""), px + 10, py + 20);
            g.drawString("AP: " + p.getActionPoints() + " | Deck: " + (deck == null ? 0 : deck), px + 10, py + 40);

            
            StarCard[] field = p.getField();
            if (field == null) continue;

            int startX = (i == 0 || i == 1) ? px + 150 : px;
            int startY = (i == 0 || i == 1) ? py : py + 70;

            for (int slot = 0; slot < field.length; slot++) {
                int sx = (i == 0 || i == 1) ? startX + (slot * 55) : startX + (slot % 3 * 45);
                int sy = (i == 0 || i == 1) ? startY : startY + (slot / 3 * 60);
                int sw = (i == 0 || i == 1) ? 50 : 40;
                int sh = (i == 0 || i == 1) ? 60 : 55;

                Rectangle sRect = new Rectangle(sx, sy, sw, sh);
                g.setColor(Color.WHITE);
                g.fillRect(sx, sy, sw, sh);
                g.setColor(Color.BLACK);
                g.drawRect(sx, sy, sw, sh);

                StarCard star = field[slot];
                if (star != null) {
                    starBounds.put(star, sRect);
                    if (selectableStars.contains(star)) {
                        g.setColor(Color.YELLOW);
                        g.drawRect(sx - 2, sy - 2, sw + 4, sh + 4);
                    }

                    
                    String imgPath = ImgPath.get(star.getName());
                    Image cardImg = null;
                    if (imgPath != null) {
                        cardImg = new ImageIcon(imgPath).getImage();
                    }

                    if (cardImg != null && cardImg.getWidth(null) > 0) {
                        g.drawImage(cardImg, sx, sy, sw, sh, this);
                    } else {
                        g.setColor(Color.WHITE);
                        g.fillRect(sx, sy, sw, sh);
                    }

                    g.setColor(Color.BLACK);
                    g.drawRect(sx, sy, sw, sh);

                    
                    g.setColor(Color.RED);
                    g.setFont(new Font("Tahoma", Font.BOLD, 10));
                    g.drawString("HP:" + star.getHp(), sx + 4, sy + 15);

                    String status = (p.isShielded(star) ? "[S]" : "") + (p.isCloaked(star) ? "[C]" : "");
                    if (!status.isEmpty()) {
                        g.setColor(Color.MAGENTA);
                        g.drawString(status, sx + 4, sy + 30);
                    }
                }
            }
        }
    }
}

import java.util.HashSet;
import java.util.List;
import java.util.Set;

public class CardActionHandler {

    
    public static String validateSpecialCard(SpecialCard special, Player myPlayer, GameState gameState, int otherHandCount) {
        String name = special.getName();

        if (name.equals("Pulsar")) {
            if (myPlayer.isPulsarUsedThisTurn()) {
                return "Pulsar ใช้ได้ 1 ครั้งต่อเทิร์นเท่านั้น";
            }
            if (otherHandCount == 0) {
                return "Pulsar: ต้องมีการ์ดอีก 1 ใบในมือเพื่อทิ้งเป็นค่าการใช้";
            }
        } else if (name.equals("Black Hole")) {
            if (gameState.isBlackHoleUsed()) {
                return "Black Hole ถูกใช้ไปแล้วในเกมนี้";
            }
        } else if (name.equals("Worm Hole")) {
            if (otherHandCount == 0) {
                return "Worm Hole: ต้องมีการ์ดอีก 1 ใบในมือเพื่อใช้แลก";
            }
        }
        return null; 
    }

    
    public static Set<StarCard> getTargetableEnemyStars(List<Player> players, Player myPlayer) {
        Set<StarCard> result = new HashSet<>();
        if (players == null) return result;

        for (Player p : players) {
            if (p == null || p == myPlayer || !p.isActive()) continue;
            for (StarCard star : p.getField()) {
                if (star != null && !p.isCloaked(star)) {
                    result.add(star);
                }
            }
        }
        return result;
    }

    
    public static Set<StarCard> getTargetableOwnStars(Player myPlayer) {
        Set<StarCard> result = new HashSet<>();
        if (myPlayer == null) return result;

        for (StarCard star : myPlayer.getField()) {
            if (star != null) {
                result.add(star);
            }
        }
        return result;
    }

    
    public static String buildServerCommand(Card card, Player target, StarCard targetStar, Card cost, List<Player> players) {
        if (card == null || card.getId() == null) {
            return null;
        }

        int targetIndex = -1;
        if (target != null && players != null) {
            targetIndex = players.indexOf(target);
        }

        int targetSlot = targetStar == null ? -1 : targetStar.getOrbitPosition();
        String costId = (cost == null || cost.getId() == null) ? "" : cost.getId();
        String command = "PLAY";

        if (card instanceof SpecialCard) {
            String category = ((SpecialCard) card).getCategory();
            if (SpecialCard.CATEGORY_ATTACK.equals(category)) {
                command = "ATTACK";
            } else if (SpecialCard.CATEGORY_DEFENSE.equals(category)) {
                command = "DEFEND";
            }
        }

        return command + "#" + card.getId() + "#" + targetIndex + "#" + targetSlot + "#" + costId;
    }

public class CardButton extends JComponent {

    private final Card card;
    private final Image image;
    private final boolean faceDown;

    private boolean hovered;
    private boolean selected;
    private Runnable onClick;

    public CardButton(Card card, Image image, boolean faceDown) {
        this.card = card;
        this.image = image;
        this.faceDown = faceDown;
        this.hovered = false;
        this.selected = false;

        setOpaque(false);
        setCursor(Cursor.getPredefinedCursor(Cursor.HAND_CURSOR));

        addMouseListener(new MouseAdapter() {
            @Override
            public void mouseEntered(MouseEvent e) {
                hovered = true;
                repaint();
            }

            @Override
            public void mouseExited(MouseEvent e) {
                hovered = false;
                repaint();
            }

            @Override
            public void mouseClicked(MouseEvent e) {
                if (onClick != null) {
                    onClick.run();
                }
            }
        });
    }

    public CardButton(Card card, Image image) {
        this(card, image, false);
    }

    public static CardButton back() {
        return new CardButton(null, null, true);
    }

    public Card getCard() {
        return card;
    }

    public Image getImage() {
        return image;
    }

    public void setOnClick(Runnable onClick) {
        this.onClick = onClick;
    }

    public void setSelected(boolean selected) {
        this.selected = selected;
        repaint();
    }

    @Override
    protected void paintComponent(Graphics g) {
        Graphics g2 = g.create();

        int w = getWidth();
        int h = getHeight();
        int pad = 3;
        int cw = w - pad * 2;
        int ch = h - pad * 2;

        if (faceDown) {
            drawCardBack(g2, pad, pad, cw, ch);
        } else if (image != null) {
            g2.drawImage(image, pad, pad, cw, ch, this);
        } else {
            drawPlaceholder(g2, pad, pad, cw, ch);
        }

        if (hovered || selected) {
            g2.setColor(selected ? new Color(255, 170, 40) : new Color(255, 90, 90));
            g2.drawRect(pad, pad, cw - 1, ch - 1);
        }

        g2.dispose();
    }

    private void drawCardBack(Graphics g2, int x, int y, int w, int h) {
        g2.setColor(new Color(255, 253, 250));
        g2.fillRect(x, y, w, h);

        g2.setColor(new Color(205, 60, 60));
        g2.drawRect(x + 2, y + 2, w - 5, h - 5);

        if (w >= 50) {
            String text = "HORIZON";
            int size = Math.max(9, Math.min(w / 7, h / 8));
            g2.setFont(new Font("Comic Sans MS", Font.BOLD, size));
            FontMetrics fm = g2.getFontMetrics();

            int tx = x + (w - fm.stringWidth(text)) / 2;
            int ty = y + (h - fm.getHeight()) / 2 + fm.getAscent();

            g2.drawString(text, tx, ty);
        } else {
            String text = "HORIZON";
            int size = Math.max(7, Math.min(w / 4, h / 9));
            g2.setFont(new Font("Comic Sans MS", Font.BOLD, size));
            FontMetrics fm = g2.getFontMetrics();

            int step = Math.max(8, h / 8);
            int total = step * text.length();
            int startY = y + (h - total) / 2 + fm.getAscent();

            for (int i = 0; i < text.length(); i++) {
                String ch = String.valueOf(text.charAt(i));
                int tx = x + (w - fm.stringWidth(ch)) / 2;
                g2.drawString(ch, tx, startY + i * step);
            }
        }
    }

    private void drawPlaceholder(Graphics g2, int x, int y, int w, int h) {
        g2.setColor(new Color(255, 253, 250));
        g2.fillRect(x, y, w, h);

        g2.setColor(new Color(205, 60, 60));
        g2.drawRect(x + 1, y + 1, w - 3, h - 3);

        String text = card == null ? "CARD" : card.getName();
        int size = Math.max(9, Math.min(w / 8, 13));
        g2.setFont(new Font("Comic Sans MS", Font.BOLD, size));
        FontMetrics fm = g2.getFontMetrics();

        int tx = x + (w - fm.stringWidth(text)) / 2;
        int ty = y + h / 2 + fm.getAscent() / 2;

        g2.drawString(text, tx, ty);
    }
}

import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.OutputStreamWriter;
import java.io.PrintWriter;
import java.net.Socket;
import java.nio.charset.StandardCharsets;

public class ClientHandler extends Thread {

    private final Socket socket;
    private final ServerContext context;
    private final BufferedReader in;
    private final PrintWriter out;
    private volatile boolean running;
    private boolean removed;
    private Player player;

    
    public ClientHandler(Socket socket, ServerContext context) throws IOException {
        if (socket == null || context == null) {
            throw new IllegalArgumentException("socket and context are required");
        }

        this.socket = socket;
        this.context = context;
        this.in = new BufferedReader(new InputStreamReader(socket.getInputStream(), StandardCharsets.UTF_8));
        this.out = new PrintWriter(new OutputStreamWriter(socket.getOutputStream(), StandardCharsets.UTF_8), true);
        this.running = true;
        this.removed = false;
        this.player = null;
    }

    @Override
    public void run() {
        try {
            String message;
            while (running && (message = in.readLine()) != null) {
                message = message.trim();
                if (message.isEmpty()) {
                    continue;
                }

                try {
                    
                    context.processClientAction(message, this);
                } catch (RuntimeException e) {
                    sendMessage("ERROR#SERVER_ERROR#Unexpected server error");
                }
            }
        } catch (IOException e) {
        } finally {
            running = false;
            closeSocket();
            notifyRemoved();
        }
    }

    public synchronized void sendMessage(String message) {
        if (message != null && !socket.isClosed()) {
            out.println(message);
        }
    }

    public void disconnect() {
        running = false;
        closeSocket();
    }

    public boolean isConnected() {
        return running && !socket.isClosed();
    }

    public Player getPlayer() {
        return player;
    }

    public void setPlayer(Player player) {
        this.player = player;
    }

    public Socket getSocket() {
        return socket;
    }

    private void notifyRemoved() {
        synchronized (this) {
            if (removed) {
                return;
            }
            removed = true;
        }
        context.removeClient(this);
    }

    private void closeSocket() {
        try {
            socket.close();
        } catch (IOException ignored) {}
    }
}

import java.util.*;

public class Deck {

    for (int i = 1; i <= 3; i++)
      cards.push(new NebulaCloak("SP-CLOAK-" + i, ImgPath.nebular_cloak));
    for (int i = 1; i <= 3; i++)
      cards.push(new StellarRegeneration("SP-REGEN-" + i, ImgPath.stallar_regenneration));
    for (int i = 1; i <= 4; i++)
      cards.push(new SpaceProbe("SP-PROBE-" + i, ImgPath.space_prode));
    for (int i = 1; i <= 1; i++)
      cards.push(new WormHole("SP-WORM-" + i, ImgPath.worm_hole));
    for (int i = 1; i <= 1; i++)
      cards.push(new BigBang("SP-BIGBANG-" + i, ImgPath.Bigbang));

    cards.push(new BlackHole("SP-BLACKHOLE-1", ImgPath.Black_hole));
  }

  
  private void addStars(String code, String name, int hp, String image, int count, int orbit) {
    for (int i = 1; i <= count; i++) {
      cards.push(new StarCard(code + "-" + i, name, hp, image, orbit));
    }
  }

  public void shuffle(){
    if (this.cards != null && !this.cards.isEmpty())
      Collections.shuffle(this.cards);
  }

  public boolean isEmpty() {
    return cards.isEmpty();
  }

  public int size() {
    return cards.size();
  }

  public Card drawCard() {
    if (cards.isEmpty()) {
      return null;
    }
    return cards.pop();
  }

  public Card searchAndRemove(String name) {
    if (name == null) {
      return null;
    }
    for (int i = 0; i < cards.size(); i++) {
      if (name.equals(cards.get(i).getName())) {
        return cards.remove(i);
      }
    }
    return null;
  }
}

    private void createRoom() {
        String username = nameField.getText().trim();
        if (username.isEmpty()) {
            return;
        }

        String portText = JOptionPane.showInputDialog(
                this,
                "Port to host on:",
                String.valueOf(GameServer.DEFAULT_PORT));

        if (portText == null) {
            return;
        }

        int port = parsePort(portText);
        if (port < 0) {
            JOptionPane.showMessageDialog(this, "Invalid port (use 1024-65535)");
            return;
        }

        GameServer server = new GameServer();
        try {
            server.startInBackground(port);
        } catch (IOException e) {
            JOptionPane.showMessageDialog(this, "Cannot create room: port " + port + " is already in use");
            return;
        }

        openWaitingRoom(true, "127.0.0.1", port, username, getLocalIp());
    }

    private void joinRoom() {
        String username = nameField.getText().trim();
        if (username.isEmpty()) {
            return;
        }

        JTextField ipField = new JTextField("127.0.0.1");
        JTextField portField = new JTextField(String.valueOf(GameServer.DEFAULT_PORT));

        JPanel panel = new JPanel(new GridLayout(2, 2, 5, 5));
        panel.add(new JLabel("Room IP:"));
        panel.add(ipField);
        panel.add(new JLabel("Port:"));
        panel.add(portField);

        int result = JOptionPane.showConfirmDialog(this, panel, "Join Room", JOptionPane.OK_CANCEL_OPTION);
        if (result != JOptionPane.OK_OPTION) {
            return;
        }

        String ip = ipField.getText().trim();
        int port = parsePort(portField.getText().trim());

        if (ip.isEmpty()) {
            JOptionPane.showMessageDialog(this, "Please enter the room IP");
            return;
        }
        if (port < 0) {
            JOptionPane.showMessageDialog(this, "Invalid port (use 1024-65535)");
            return;
        }

        openWaitingRoom(false, ip, port, username, ip);
    }

    private void openWaitingRoom(boolean isHost, String connectIp, int port, String username, String displayIp) {
        GameClient client = new GameClient();
        WaitingRoomWindow room = new WaitingRoomWindow(client, isHost, displayIp, port, username);
        client.setOnMessage(room::handleMessage);

        if (!client.connect(connectIp, port, username)) {
            room.dispose();
            JOptionPane.showMessageDialog(this, "Cannot connect to room: " + connectIp + ":" + port);
            return;
        }

        room.setVisible(true);
        dispose();
    }

    private int parsePort(String text) {
        try {
            int port = Integer.parseInt(text.trim());
            if (port >= 1024 && port <= 65535) {
                return port;
            }
        } catch (NumberFormatException e) {
            return -1;
        }
        return -1;
    }

    private String getLocalIp() {
        try {
            String ip = InetAddress.getLocalHost().getHostAddress();
            return ip.contains(":") ? "127.0.0.1" : ip;
        } catch (UnknownHostException e) {
            return "127.0.0.1";
        }
    }

                }
            } catch (IOException e) {
            }

        listener.start();

        }
    }

        }
    }

import java.io.IOException;
import java.io.PrintWriter;
import java.net.ServerSocket;
import java.net.Socket;
import java.util.ArrayList;
import java.util.List;
import java.util.concurrent.CopyOnWriteArrayList;

public class GameServer implements ServerContext {

    private static final int MAX_PLAYERS = 4;
    private static final int OPENING_HAND = 5;
    public static final int DEFAULT_PORT = 5000;
    private ServerSocket serverSocket;
    private List<ClientHandler> clients;
    private GameState gameState;

    public GameServer() {
        this.clients = new CopyOnWriteArrayList<>();
        this.gameState = null;
    }

    public void startInBackground(int port) throws IOException {
        this.serverSocket = new ServerSocket(port);

        Thread serverThread = new Thread(this::acceptLoop);
        serverThread.setDaemon(true);
        serverThread.start();
    }

    private void acceptLoop() {
        try {
            while (true) {
                Socket socket = serverSocket.accept();

                if (clients.size() >= MAX_PLAYERS) {
                    PrintWriter out = new PrintWriter(socket.getOutputStream(), true);
                    out.println("ERR#SERVER_FULL");
                    socket.close();
                    continue;
                }

                ClientHandler handler = new ClientHandler(socket, this);
                synchronized (this) {
                    clients.add(handler);
                }
                handler.start();
            }
        } catch (IOException e) {
        }
    }

    public synchronized void broadcast(String message) {
        for (ClientHandler client : clients) {
            client.sendMessage(message);
        }
    }

    @Override
    public synchronized void processClientAction(String action, ClientHandler sender) {
        if (action == null || action.isEmpty()) {
            return;
        }

        String[] parts = action.split("#", -1);
        String command = parts[0];

        switch (command) {
            case "JOIN": {
                if (gameState != null) {
                    sender.sendMessage("ERR#ALREADY_STARTED");
                    break;
                }

                if (parts.length < 2) {
                    sender.sendMessage("ERR#INVALID_NAME");
                    break;
                }

                String username = parts[1].trim();
                if (username.isEmpty()) {
                    sender.sendMessage("ERR#INVALID_NAME");
                    break;
                }

                if (hasPlayerName(username)) {
                    sender.sendMessage("ERR#NAME_TAKEN");
                    break;
                }

                sender.setPlayer(new Player(username));
                broadcastLobby();
                break;
            }

            case "PLAY":
            case "ATTACK":
            case "DEFEND":
                handleCardAction(sender, parts);
                break;

            case "START": {
                if (gameState != null) {
                    sender.sendMessage("ERR#ALREADY_STARTED");
                    break;
                }
                if (!isHost(sender)) {
                    sender.sendMessage("ERR#NOT_HOST");
                    break;
                }
                if (countPlayers() < 2) {
                    sender.sendMessage("ERR#NEED_MORE_PLAYERS");
                    break;
                }

                List<Player> players = buildPlayerList();
                dealOpeningHands(players);
                gameState = new GameState(players);
                startTurn(gameState.getCurrentPlayer());
                gameState.addChangeListener(this::broadcastState);

                StringBuilder start = new StringBuilder("SYSTEM#GAME_START");

                for (Player player : players) {
                    start.append("#").append(player.getUsername());
                }

                broadcast(start.toString());
                broadcastState();
                break;
            }

            case "END_TURN":
                if (gameState == null) {
                    sender.sendMessage("ERR#GAME_NOT_READY");
                    break;
                }
                if (gameState.isGameOver()) {
                    sender.sendMessage("ERR#GAME_OVER");
                    break;
                }
                if (sender.getPlayer() != gameState.getCurrentPlayer()) {
                    sender.sendMessage("ERR#NOT_YOUR_TURN");
                    break;
                }

                Player current = gameState.getCurrentPlayer();

                gameState.setCurrentPhase(3);

                if (gameState.checkWinCondition(current)) {
                    break;
                }

                enforceHandLimit(current);

                gameState.nextTurn();
                startTurn(gameState.getCurrentPlayer());
                broadcastState();
                break;

            default:
                sender.sendMessage("ERR#UNKNOWN_COMMAND");
                break;
        }
    }

    private void handleCardAction(ClientHandler sender, String[] parts) {
        if (gameState == null) {
            sendError(sender, "GAME_NOT_READY");
            return;
        }

        if (gameState.isGameOver()) {
            sendError(sender, "GAME_OVER");
            return;
        }

        Player owner = sender.getPlayer();

        if (owner == null) {
            sendError(sender, "INVALID_PLAYER");
            return;
        }

        if (!owner.isActive()) {
            sendError(sender, "PLAYER_INACTIVE");
            return;
        }

        if (owner != gameState.getCurrentPlayer()) {
            sendError(sender, "NOT_YOUR_TURN");
            return;
        }

        if (parts.length < 2 || parts[1].trim().isEmpty()) {
            sendError(sender, "INVALID_ACTION");
            return;
        }

        Card card = findCardInHand(owner, parts[1].trim());

        if (card == null) {
            sendError(sender, "CARD_NOT_IN_HAND");
            return;
        }

        int targetIndex = parseInt(parts.length > 2 ? parts[2] : "", -1);
        int targetSlot = parseInt(parts.length > 3 ? parts[3] : "", -1);
        String costId = parts.length > 4 ? parts[4].trim() : "";

        Player target = null;

        if (targetIndex != -1) {
            if (targetIndex < 0 || targetIndex >= gameState.getPlayers().size()) {
                sendError(sender, "INVALID_TARGET");
                return;
            }

            target = gameState.getPlayers().get(targetIndex);

            if (target == null) {
                sendError(sender, "INVALID_TARGET");
                return;
            }
        }

        StarCard targetStar = null;

        if (targetSlot != -1) {
            if (target == null || targetSlot < 0 || targetSlot >= target.getField().length) {
                sendError(sender, "INVALID_TARGET");
                return;
            }

            targetStar = target.getField()[targetSlot];

            if (targetStar == null) {
                sendError(sender, "INVALID_TARGET");
                return;
            }
        }

        Card cost = null;

        if (!costId.isEmpty()) {
            cost = findCardInHand(owner, costId);

            if (cost == null || cost == card) {
                sendError(sender, "INVALID_COST");
                return;
            }
        }

        if (card instanceof StarCard) {
            playStar(sender, owner, (StarCard) card);
        } else if (card instanceof SpecialCard) {
            playSpecial(sender, owner, (SpecialCard) card, target, targetStar, cost);
        } else {
            sendError(sender, "UNSUPPORTED_CARD");
        }
    }

    private void playStar(ClientHandler sender, Player owner, StarCard star) {
        if (owner.hasStarOnField(star.getName())) {
            sendError(sender, "DUPLICATE_STAR");
            return;
        }

        if (owner.getActionPoints() < 1) {
            sendError(sender, "NOT_ENOUGH_AP");
            return;
        }

        int pos = star.getOrbitPosition();

        if (pos < 0 || pos >= owner.getField().length || owner.getField()[pos] != null) {
            pos = owner.findFirstEmptySlot();
        }

        if (pos < 0) {
            sendError(sender, "FIELD_FULL");
            return;
        }

        star.setOrbitPosition(pos);

        if (!owner.placeStar(star)) {
            sendError(sender, "CANNOT_PLAY_CARD");
            return;
        }

        owner.removeCardFromHand(star);
        owner.useActionPoint(1);
        gameState.checkWinCondition(owner);
        broadcastState();
    }

    private void playSpecial(ClientHandler sender, Player owner, SpecialCard card,
                             Player target, StarCard targetStar, Card cost) {
        String name = card.getName();
        String targetType = card.getTargetType();

        if (target != null && !target.isActive()) {
            sendError(sender, "INVALID_TARGET");
            return;
        }

        
        if (name.equals("Pulsar")) {
            if (owner.isPulsarUsedThisTurn()) {
                sendError(sender, "PULSAR_ALREADY_USED");
                return;
            }
            if (cost == null) {
                sendError(sender, "INVALID_COST");
                return;
            }
        } else if (name.equals("Black Hole")) {
            if (gameState.isBlackHoleUsed()) {
                sendError(sender, "BLACK_HOLE_ALREADY_USED");
                return;
            }
        } else if (name.equals("Worm Hole")) {
            if (cost == null || target == null || target.getHand().isEmpty()) {
                sendError(sender, "INVALID_TARGET");
                return;
            }
        }

        
        if (SpecialCard.TARGET_ENEMY_STAR.equals(targetType)) {
            if (target == null || target == owner || targetStar == null || target.isCloaked(targetStar)) {
                sendError(sender, "INVALID_TARGET");
                return;
            }
        } else if (SpecialCard.TARGET_ALLY_STAR.equals(targetType)) {
            if (target != owner || targetStar == null) {
                sendError(sender, "INVALID_TARGET");
                return;
            }
        } else if (SpecialCard.TARGET_ENEMY_PLAYER.equals(targetType)) {
            if (target == null || target == owner) {
                sendError(sender, "INVALID_TARGET");
                return;
            }
        }

        if (!spendActionPoint(sender, owner)) {
            return;
        }

        if (name.equals("Pulsar")) {
            ((Pulsar) card).playEffect(owner, target, targetStar, cost, gameState);
        } else if (name.equals("Worm Hole")) {
            ((WormHole) card).playEffectWithCard(owner, target, cost, gameState);
        } else {
            card.playEffect(owner, target, targetStar, gameState);
        }

        finishSpecial(owner, card);
    }

    private boolean spendActionPoint(ClientHandler sender, Player owner) {
        if (owner.getActionPoints() < 1) {
            sendError(sender, "NOT_ENOUGH_AP");
            return false;
        }

        return owner.useActionPoint(1);
    }

    private void finishSpecial(Player owner, SpecialCard card) {
        owner.removeCardFromHand(card);

        if (!card.getName().equals("Black Hole")) {
            owner.addToDropZone(card);
        }

        gameState.checkWinCondition(owner);
        broadcastState();
    }

    private Card findCardInHand(Player player, String cardId) {
        for (Card card : player.getHand()) {
            if (card.getId() != null && card.getId().equals(cardId)) {
                return card;
            }
        }

        return null;
    }

    private void sendError(ClientHandler sender, String code) {
        sender.sendMessage("ERR#" + code);
    }

    private static int parseInt(String value, int fallback) {
        try {
            return Integer.parseInt(value.trim());
        } catch (NumberFormatException e) {
            return fallback;
        }
    }

    private void dealOpeningHands(List<Player> players) {
        for (int round = 0; round < OPENING_HAND; round++) {
            for (Player player : players) {
                Card card = player.getDeck().drawCard();

                if (card != null) {
                    player.addCardToHand(card);
                }
            }
        }
    }

    private boolean startTurn(Player player) {
        if (player == null || gameState == null) {
            return false;
        }

        gameState.setCurrentPhase(1);
        player.clearExpiredBuffs();

        Card card = player.getDeck().drawCard();

        if (card != null) {
            player.addCardToHand(card);
            gameState.setCurrentPhase(2);
            return true;
        }

        player.setActive(false);

        if (gameState.getActivePlayerCount() <= 1) {
            gameState.setWinner(gameState.getFirstActivePlayer());
            gameState.setGameOver(true);
            return false;
        }

        gameState.nextTurn();
        return startTurn(gameState.getCurrentPlayer());
    }

    private void enforceHandLimit(Player player) {
        if (player == null) {
            return;
        }

        int discarded = 0;

        while (player.getHand().size() > Player.MAX_HAND) {
            Card card = player.getHand().get(player.getHand().size() - 1);
            player.removeCardFromHand(card);
            player.addToDropZone(card);
            discarded++;
        }

        if (discarded > 0) {
            sendToPlayer(player, "SYSTEM#มือเกิน 7 ใบ: ทิ้ง " + discarded
                    + " ใบลง Drop Zone");
        }
    }

    private void sendToPlayer(Player player, String message) {
        for (ClientHandler client : clients) {
            if (client.getPlayer() == player) {
                client.sendMessage(message);
                return;
            }
        }
    }

    @Override
    public synchronized void removeClient(ClientHandler handler) {
        clients.remove(handler);

        Player player = handler.getPlayer();

        if (player != null) {
            broadcast("SYSTEM#" + player.getUsername() + "_LEFT");

            if (gameState != null && gameState.getPlayers().contains(player)) {
                player.setActive(false);

                if (gameState.getActivePlayerCount() <= 1) {
                    gameState.setWinner(gameState.getFirstActivePlayer());
                    gameState.setGameOver(true);
                } else if (gameState.getCurrentPlayer() == player) {
                    gameState.nextTurn();
                    startTurn(gameState.getCurrentPlayer());
                }

                broadcastState();
            }
        }

        if (gameState == null) {
            broadcastLobby();
        }
    }

    private void broadcastState() {
        if (gameState == null) {
            return;
        }

        String snapshot = gameState.toProtocolString();

        for (ClientHandler client : clients) {
            Player player = client.getPlayer();

            if (player == null) {
                continue;
            }

            client.sendMessage(snapshot);
            client.sendMessage(toHandMessage(player));
        }
    }

    private String toHandMessage(Player player) {
        StringBuilder sb = new StringBuilder("HAND");

        for (Card card : player.getHand()) {
            sb.append("#").append(safeToken(card.getId())).append(",")
                    .append(safeToken(card.getName())).append(",")
                    .append(safeToken(card.getType())).append(",");

            if (card instanceof StarCard) {
                StarCard star = (StarCard) card;
                sb.append(star.getMaxHp()).append(",")
                        .append(star.getOrbitPosition());
            } else {
                sb.append("0,-1");
            }
        }

        return sb.toString();
    }

    private static String safeToken(String value) {
        if (value == null) {
            return "";
        }

        return value.replace(',', ' ').replace('#', ' ').replace(':', ' ');
    }

    private void broadcastLobby() {
        StringBuilder lobby = new StringBuilder("LOBBY");

        for (ClientHandler client : clients) {
            Player player = client.getPlayer();

            if (player != null) {
                lobby.append("#").append(player.getUsername());
            }
        }

        broadcast(lobby.toString());
    }

    private boolean hasPlayerName(String username) {
        for (ClientHandler client : clients) {
            Player player = client.getPlayer();

            if (player != null && player.getUsername().equals(username)) {
                return true;
            }
        }
        return false;
    }

    private boolean isHost(ClientHandler sender) {
        for (ClientHandler client : clients) {
            if (client.getPlayer() != null) {
                return client == sender;
            }
        }
        return false;
    }

    private int countPlayers() {
        int count = 0;

        for (ClientHandler client : clients) {
            if (client.getPlayer() != null) {
                count++;
            }
        }

        return count;
    }

    private List<Player> buildPlayerList() {
        List<Player> players = new ArrayList<>();

        for (ClientHandler client : clients) {
            if (client.getPlayer() != null) {
                players.add(client.getPlayer());
            }
        }

        return players;
    }

import java.util.ArrayList;
import java.util.List;

public class GameState {
    private List<Player> players;
    private int currentPlayerIndex;
    private int currentPhase;
    private boolean isGameOver;
    private Player winner;
    private boolean blackHoleUsed;
    private final List<Runnable> changeListeners;

    public GameState(List<Player> players) {
        this.players = players;
        this.currentPlayerIndex = 0;
        this.currentPhase = 1;
        this.isGameOver = false;
        this.winner = null;
        this.blackHoleUsed = false;
        this.changeListeners = new ArrayList<>();
        bindPlayerListeners();
    }

    private void bindPlayerListeners() {
        if (players == null) {
            return;
        }

        for (Player player : players) {
            if (player != null) {
                player.setChangeListener(this::fireStateChanged);
            }
        }
    }

    public void addChangeListener(Runnable listener) {
        if (listener != null && !changeListeners.contains(listener)) {
            changeListeners.add(listener);
        }
    }

    public void removeChangeListener(Runnable listener) {
        changeListeners.remove(listener);
    }

    private void fireStateChanged() {
        for (Runnable listener : new ArrayList<>(changeListeners)) {
            listener.run();
        }
    }

    public Player getCurrentPlayer() {
        if (players == null || players.isEmpty()) return null;
        return players.get(currentPlayerIndex);
    }

    public void nextTurn() {
        if (isGameOver || players == null || players.isEmpty()) return;

        
        for (int step = 0; step < players.size(); step++) {
            currentPlayerIndex = (currentPlayerIndex + 1) % players.size();
            Player nextPlayer = getCurrentPlayer();

            if (nextPlayer != null && nextPlayer.isActive()) {
                currentPhase = 1;
                nextPlayer.resetActionPoints();
                fireStateChanged();
                return;
            }
        }
    }

    public int getActivePlayerCount() {
        if (players == null) {
            return 0;
        }

        int count = 0;

        for (Player player : players) {
            if (player != null && player.isActive()) {
                count++;
            }
        }

        return count;
    }

    public Player getFirstActivePlayer() {
        if (players == null) {
            return null;
        }

        for (Player player : players) {
            if (player != null && player.isActive()) {
                return player;
            }
        }

        return null;
    }


    public void advancePhase() {
        if (currentPhase < 3)
            currentPhase++;
        else
            nextTurn();

    }

    public boolean checkWinCondition(Player player) {
        if (player != null && player.checkWinCondition()) {
            this.isGameOver = true;
            this.winner = player;
            fireStateChanged();
            return true;
        }
        return false;
    }

    public void eliminatePlayer(Player player) {
        if (player == null || !players.remove(player)) {
            return;
        }

        if (players.size() == 1) {
            this.isGameOver = true;
            this.winner = players.get(0);
        } else if (currentPlayerIndex >= players.size()) {
            currentPlayerIndex = 0;
        }

        fireStateChanged();
    }

    public List<Player> getPlayers() {
        return players;
    }

    public int getCurrentPlayerIndex() {
        return currentPlayerIndex;
    }

    public int getCurrentPhase() {
        return currentPhase;
    }

    public boolean isGameOver() {
        return isGameOver;
    }

    public Player getWinner() {
        return winner;
    }

    public boolean isBlackHoleUsed() {
        return blackHoleUsed;
    }

    public void setBlackHoleUsed(boolean blackHoleUsed) {
        if (this.blackHoleUsed != blackHoleUsed) {
            this.blackHoleUsed = blackHoleUsed;
            fireStateChanged();
        }
    }

    public void setCurrentPlayerIndex(int currentPlayerIndex) {
        if (players == null || players.isEmpty()) {
            this.currentPlayerIndex = 0;
        } else {
            this.currentPlayerIndex = Math.max(0,
                    Math.min(currentPlayerIndex, players.size() - 1));
        }
    }

    public void setCurrentPhase(int currentPhase) {
        this.currentPhase = Math.max(1, Math.min(3, currentPhase));

    public void setWinner(Player winner) {
        this.winner = winner;

        if (winner != null) {

    public String toProtocolString() {
        StringBuilder sb = new StringBuilder("UPDATE_BOARD");
        sb.append("#TURN:").append(currentPlayerIndex);
        sb.append("#PHASE:").append(currentPhase);
        sb.append("#BH:").append(blackHoleUsed ? "1" : "0");
        sb.append("#OVER:").append(isGameOver ? "1" : "0");
        sb.append("#WINNER:").append(players == null ? -1 : players.indexOf(winner));

        if (players == null) {
            return sb.toString();
        }

        for (int i = 0; i < players.size(); i++) {
            Player player = players.get(i);

            if (player == null) {
                continue;
            }

            sb.append("#P:").append(i).append(",")
                    .append(safeToken(player.getUsername())).append(",")
                    .append(player.getActionPoints()).append(",")
                    .append(player.getHand().size()).append(",")
                    .append(player.getDeck().size()).append(",")
                    .append(player.getDropZone().size()).append(",")
                    .append(player.isActive() ? "1" : "0");

            StarCard[] field = player.getField();

            for (int slot = 0; slot < field.length; slot++) {
                StarCard star = field[slot];

                if (star == null) {
                    continue;
                }

                sb.append("#F:").append(i).append(",")
                        .append(slot).append(",")
                        .append(safeToken(star.getName())).append(",")
                        .append(star.getHp()).append(",")
                        .append(star.getMaxHp()).append(",")
                        .append(player.isShielded(star) ? "1" : "0").append(",")
                        .append(player.isCloaked(star) ? "1" : "0");
            }

            for (Card card : player.getDropZone()) {
                sb.append("#D:").append(i).append(",")
                        .append(safeToken(card.getId())).append(",")
                        .append(safeToken(card.getName())).append(",")
                        .append(safeToken(card.getType())).append(",");

                if (card instanceof StarCard) {
                    StarCard star = (StarCard) card;
                    sb.append(star.getMaxHp()).append(",")
                            .append(star.getOrbitPosition());
                } else {
                    sb.append("0,-1");
                }
            }
        }

        return sb.toString();
    }

    private static String safeToken(String value) {
        if (value == null) {
            return "";
        }

        return value.replace(',', ' ').replace('#', ' ').replace(':', ' ');
    }

    private void startCardPlay(Card card) {
        if (myPlayer == null || gameState == null || players == null) {
            setStatus("ยังไม่มีเกมที่พร้อมเล่น");
            return;
        }

        if (gameState.isGameOver()) {
            setStatus("เกมจบแล้ว");
            return;
        }

        if (!myPlayer.isActive()) {
            setStatus("คุณแพ้แล้ว — ออกจากเกม (Deck Out)");
            return;
        }

        if (client != null && gameState.getCurrentPlayer() != myPlayer) {
            setStatus("ยังไม่ถึงเทิร์นของคุณ");
            return;
        }

        if (!myPlayer.getHand().contains(card)) {
            return;
        }

        if (myPlayer.getActionPoints() <= 0) {
            setStatus("AP หมดแล้ว (0/2) — กด \"จบเทิร์น\" เพื่อเริ่มเทิร์นใหม่");
            updateMyHand(myPlayer.getHand());
            return;
        }

        if (card instanceof StarCard) {
            sendPlayToServer(card, null, null, null);
            return;
        }

        if (!(card instanceof SpecialCard)) {
            setStatus("ไม่รู้จักชนิดการ์ดนี้: " + card.getName());
            return;
        }

        SpecialCard special = (SpecialCard) card;
        String error = CardActionHandler.validateSpecialCard(special, myPlayer, gameState, otherHandCardCount(card));
        if (error != null) {
            setStatus(error);
            return;
        }

        String targetType = special.getTargetType();
        String prompt = special.getPromptMessage();

        if (SpecialCard.TARGET_ENEMY_STAR.equals(targetType)) {
            Set<StarCard> targets = CardActionHandler.getTargetableEnemyStars(players, myPlayer);
            if (targets.isEmpty()) {
                setStatus(special.getName() + ": ไม่มีดาวคู่แข่งที่เลือกได้");
                return;
            }
            beginTargeting(card, TargetingManager.PendingMode.ENEMY_STAR, targets, null, prompt);
        } else if (SpecialCard.TARGET_ALLY_STAR.equals(targetType)) {
            Set<StarCard> targets = CardActionHandler.getTargetableOwnStars(myPlayer);
            if (targets.isEmpty()) {
                setStatus(special.getName() + ": คุณยังไม่มีดาวบนสนาม");
                return;
            }
            beginTargeting(card, TargetingManager.PendingMode.ALLY_STAR, targets, null, prompt);
        } else if (SpecialCard.TARGET_ENEMY_PLAYER.equals(targetType)) {
            Set<Player> targets = new HashSet<>();
            for (Player player : players) {
                if (player != myPlayer && player.isActive()) {
                    if (special.getName().equals("Supernova")) {
                        if (hasTargetableStar(player)) targets.add(player);
                    } else if (special.getName().equals("Worm Hole")) {
                        if (handCountOf(player) > 0) targets.add(player);
                    } else {
                        targets.add(player);
                    }
                }
            }
            if (targets.isEmpty()) {
                setStatus(special.getName() + ": ไม่มีผู้เล่นเป้าหมายที่เลือกได้");
                return;
            }
            beginTargeting(card, TargetingManager.PendingMode.ENEMY_PLAYER, null, targets, prompt);
        } else if (SpecialCard.TARGET_NONE.equals(targetType)) {
            sendPlayToServer(card, null, null, null);
        } else {
            setStatus("ยังไม่ได้เชื่อม Logic ของการ์ด: " + special.getName());
        }
    }

    public void updateBoardState(String state) {

        if (state == null || state.isEmpty()) return;

        if (state.startsWith("SYSTEM#")) {
            setStatus(state.substring("SYSTEM#".length()).replace('_', ' '));
            return;
        }

        if (state.startsWith("ERR#")) {
            setStatus("เซิร์ฟเวอร์: " + state.substring("ERR#".length()).replace('#', ' '));
            return;
        }

        if (state.equals("HAND") || state.startsWith("HAND#")) {
            applyHandSnapshot(state);
            return;
        }

        if (!state.startsWith("UPDATE_BOARD#")) return;

        applyBoardSnapshot(state);
    }

    private void applyBoardSnapshot(String snapshot) {
        ProtocolParser.BoardUpdateData data = ProtocolParser.parseBoard(snapshot);

        List<Player> targetPlayers = players;
        if ((targetPlayers == null || targetPlayers.isEmpty()) && gameState != null) {
            targetPlayers = gameState.getPlayers();
        }

        if (targetPlayers == null) return;

        cancelPending();

        int[] oldDropCounts = new int[targetPlayers.size()];
        for (int i = 0; i < targetPlayers.size(); i++) {
            Player player = targetPlayers.get(i);
            if (player != null) {
                oldDropCounts[i] = player.getDropZone().size();
                player.clearField();
                player.getDropZone().clear();
            }
        }

        int[] handCountsByIndex = new int[targetPlayers.size()];
        int[] deckCountsByIndex = new int[targetPlayers.size()];
        for (int i = 0; i < deckCountsByIndex.length; i++) deckCountsByIndex[i] = -1;

        for (String record : data.playerRecords) {
            String[] parts = record.split(",", -1);
            if (parts.length < 6) continue;

            int index = parseInt(parts[0], -1);
            if (index < 0 || index >= targetPlayers.size()) continue;

            Player player = targetPlayers.get(index);
            if (player == null) continue;

            player.setActionPoints(parseInt(parts[2], player.getActionPoints()));
            player.setActive(parts.length < 7 || !"0".equals(parts[6]));
            handCountsByIndex[index] = Math.max(0, parseInt(parts[3], player.getHand().size()));
            deckCountsByIndex[index] = Math.max(0, parseInt(parts[4], -1));
        }

        deckCounts.clear();
        handCounts.clear();

        for (int i = 0; i < deckCountsByIndex.length; i++) {
            if (targetPlayers.get(i) == null) continue;
            handCounts.put(targetPlayers.get(i), handCountsByIndex[i]);
            if (deckCountsByIndex[i] >= 0) {
                deckCounts.put(targetPlayers.get(i), deckCountsByIndex[i]);
            }
        }

        boardPanel.setDeckCounts(deckCounts);
        updateDeckLabel();

        for (String record : data.fieldRecords) {
            String[] parts = record.split(",", -1);
            if (parts.length < 7) continue;

            int playerIndex = parseInt(parts[0], -1);
            int slot = parseInt(parts[1], -1);
            if (playerIndex < 0 || playerIndex >= targetPlayers.size()) continue;

            Player player = targetPlayers.get(playerIndex);
            if (player == null || slot < 0 || slot >= player.getField().length) continue;

            int maxHp = Math.max(1, parseInt(parts[4], 1));
            int hp = Math.max(0, Math.min(maxHp, parseInt(parts[3], maxHp)));
            String cardName = parts[2];
            StarCard star = new StarCard("NET-" + playerIndex + "-" + slot,
                    cardName, maxHp, ImgPath.get(cardName), slot);

            if (hp < maxHp) star.takeDamage(maxHp - hp);
            if (!player.placeStar(star)) continue;

            if ("1".equals(parts[5])) player.addShield(star);
            if ("1".equals(parts[6])) player.addCloak(star);
        }

        for (String record : data.dropRecords) {
            String[] parts = record.split(",", -1);
            if (parts.length < 6) continue;

            int playerIndex = parseInt(parts[0], -1);
            if (playerIndex < 0 || playerIndex >= targetPlayers.size()) continue;

            Player player = targetPlayers.get(playerIndex);
            if (player == null) continue;

            Card card = CardFactory.create(parts[1], parts[2], parts[3],
                    parseInt(parts[4], 0), parseInt(parts[5], -1));

            if (card != null) {
                player.getDropZone().add(card);
            }
        }

        for (int i = 0; i < targetPlayers.size(); i++) {
            Player player = targetPlayers.get(i);
            if (player == null) continue;

            int newDropCount = player.getDropZone().size();
            if (newDropCount > oldDropCounts[i] && newDropCount > 0) {
                Card lastDrop = player.getDropZone().get(newDropCount - 1);
                boardPanel.playCardToDropZone(lastDrop, loadCardImage(lastDrop));
            }
        }

        if (gameState != null) {
            if (data.turn >= 0) gameState.setCurrentPlayerIndex(data.turn);
            if (data.phase >= 0) gameState.setCurrentPhase(data.phase);

            gameState.setBlackHoleUsed(data.blackHoleUsed);
            gameState.setGameOver(data.gameOver);

            if (data.winnerIndex >= 0 && data.winnerIndex < targetPlayers.size()) {
                gameState.setWinner(targetPlayers.get(data.winnerIndex));
            }

            if (data.gameOver) {
                showGameOver(gameState.getWinner());
            } else if (gameState.getCurrentPlayer() != null) {
                Player current = gameState.getCurrentPlayer();
                String phaseText = phaseName(gameState.getCurrentPhase());

                if (current == myPlayer) {
                    setStatus("ถึงเทิร์นของคุณ — AP " + current.getActionPoints() + " (เฟส " + phaseText + ")");
                } else {
                    setStatus("รอเทิร์นของ " + current.getUsername() + " (เฟส " + phaseText + ")");
                }
            }

            if (statusBarPanel != null) {
                boolean canEndTurn = !data.gameOver && myPlayer != null && myPlayer.isActive()
                        && gameState.getCurrentPlayer() == myPlayer;
                statusBarPanel.setEndTurnEnabled(canEndTurn);
            }

            if (!data.gameOver && myPlayer != null && !myPlayer.isActive()) {
                if (statusBarPanel != null) statusBarPanel.setEndTurnEnabled(false);
                setStatus("คุณแพ้แล้ว — เด็คหมด (Deck Out)");
            }
        }

        if (myPlayer != null) {
            updateMyHand(myPlayer.getHand());
        }

        if (targetPlayers.size() >= 2) {
            int topCount = handCountsByIndex[1];
            int leftCount = targetPlayers.size() > 2 ? handCountsByIndex[2] : 0;
            int rightCount = targetPlayers.size() > 3 ? handCountsByIndex[3] : 0;

            int cardH = (int) Math.round(cardW / 1.6);

            int totalH = count * cardH + (count - 1) * gap;
            if (totalH > availableH) {
                cardH = Math.max(20, (availableH - (count - 1) * gap) / count);
                cardW = (int) Math.round(cardH * 1.6);
                totalH = count * cardH + (count - 1) * gap;
            }

            int startX = (availableW - cardW) / 2;
            int startY = Math.max(0, (availableH - totalH) / 2);
            int y = startY;

            for (Component card : getComponents()) {
                card.setBounds(startX, y, cardW, cardH);
                y += cardH + gap;

import java.util.ArrayList;
import java.util.List;

public class ProtocolParser {

    
    public static class BoardUpdateData {
        public int turn = -1;
        public int phase = -1;
        public int winnerIndex = -1;
        public boolean blackHoleUsed = false;
        public boolean gameOver = false;
        public List<String> playerRecords = new ArrayList<>();
        public List<String> fieldRecords = new ArrayList<>();
        public List<String> dropRecords = new ArrayList<>();
    }

    
    public static List<Card> parseHand(String message) {
        List<Card> newHand = new ArrayList<>();
        if (message == null || message.isEmpty()) {
            return newHand;
        }

        String body = message.startsWith("HAND#") ? message.substring("HAND#".length()) : "";

        if (!body.isEmpty()) {
            for (String record : body.split("#", -1)) {
                String[] parts = record.split(",", -1);
                if (parts.length < 5) continue;

                Card card = CardFactory.create(
                        parts[0],
                        parts[1],
                        parts[2],
                        parseInt(parts[3], 0),
                        parseInt(parts[4], -1)
                );

                if (card != null) {
                    newHand.add(card);
                }
            }
        }
        return newHand;
    }

    
    public static BoardUpdateData parseBoard(String snapshot) {
        BoardUpdateData data = new BoardUpdateData();
        if (snapshot == null || snapshot.isEmpty()) {
            return data;
        }

        String[] tokens = snapshot.split("#");

        for (int i = 1; i < tokens.length; i++) {
            String token = tokens[i];
            if (token.startsWith("TURN:")) {
                data.turn = parseInt(token.substring("TURN:".length()), -1);
            } else if (token.startsWith("PHASE:")) {
                data.phase = parseInt(token.substring("PHASE:".length()), -1);
            } else if (token.startsWith("BH:")) {
                data.blackHoleUsed = "1".equals(token.substring("BH:".length()));
            } else if (token.startsWith("OVER:")) {
                data.gameOver = "1".equals(token.substring("OVER:".length()));
            } else if (token.startsWith("WINNER:")) {
                data.winnerIndex = parseInt(token.substring("WINNER:".length()), -1);
            } else if (token.startsWith("P:")) {
                data.playerRecords.add(token.substring("P:".length()));
            } else if (token.startsWith("F:")) {
                data.fieldRecords.add(token.substring("F:".length()));
            } else if (token.startsWith("D:")) {
                data.dropRecords.add(token.substring("D:".length()));
            }
        }
        return data;
    }

    private static int parseInt(String value, int fallback) {
        try {
            return Integer.parseInt(value.trim());
        } catch (NumberFormatException e) {
            return fallback;
        }
    }
}

        return "Connected: " + displayIp + ":" + port;
    }

    public void handleMessage(String message) {
        if (message == null) {
            return;
        }

        SwingUtilities.invokeLater(() -> {
            if (message.startsWith("LOBBY#")) {
                updateLobby(message);
            } else if (message.startsWith("SYSTEM#GAME_START")) {
                startGame(message);
            } else if (message.startsWith("ERR#")) {
                JOptionPane.showMessageDialog(this, errorText(message));

                if (message.equals("ERR#SERVER_FULL")
                        || message.equals("ERR#NAME_TAKEN")
                        || message.equals("ERR#ALREADY_STARTED")) {
                    dispose();
                }
            }
        });
    }

    private void updateLobby(String message) {
        String[] parts = message.split("#");
        StringBuilder names = new StringBuilder();

        lobbyNames.clear();

        for (int i = 1; i < parts.length; i++) {
            String name = parts[i].trim();

            if (name.isEmpty()) {
                continue;
            }

            lobbyNames.add(name);
            names.append(lobbyNames.size()).append(". ").append(name).append("\n");
        }

        int playerCount = lobbyNames.size();
        namesArea.setText(names.toString());
        countLabel.setText("Players in room: " + playerCount + "/4");
        startButton.setEnabled(isHost && playerCount >= 2);
    }

    private void startGame(String message) {
        String[] parts = message.split("#", -1);
        List<String> names = new ArrayList<>();

        for (int i = 2; i < parts.length; i++) {
            String name = parts[i].trim();

            if (!name.isEmpty()) {
                names.add(name);
            }
        }

        if (names.isEmpty()) {
            names.addAll(lobbyNames);
        }

        if (names.isEmpty()) {
            JOptionPane.showMessageDialog(this, "Cannot start game: player list is empty");
            return;
        }

        List<Player> players = new ArrayList<>();
        Player me = null;

        for (String name : names) {
            Player player = new Player(name);
            players.add(player);

            if (name.equals(username)) {
                me = player;
            }
        }

        if (me == null) {
            me = players.get(0);
        }

        GameWindow window = new GameWindow();
        window.setPlayers(players);
        window.setMyPlayer(me);
        window.setGameClient(client);

        client.setUi(window);
        client.setOnMessage(window::updateBoardState);

        window.setVisible(true);
        dispose();
    }

    private String errorText(String message) {
        if (message.equals("ERR#SERVER_FULL")) {
            return "Room is full (max 4 players)";
        }
        if (message.equals("ERR#NAME_TAKEN")) {
            return "That name is already taken";
        }
        if (message.equals("ERR#NOT_HOST")) {
            return "Only the host can start the game";
        }
        if (message.equals("ERR#NEED_MORE_PLAYERS")) {
            return "Need at least 2 players";
        }
        if (message.equals("ERR#ALREADY_STARTED")) {
            return "The game has already started";
        }
        if (message.equals("ERR#INVALID_NAME")) {
            return "Invalid player name";
        }
        return message;
    }
}

    public BigBang(String id, String imagePath)
    {
        super(id, "Big Bang", SpecialCard.CATEGORY_SECRET, imagePath);

    }

    @Override
    public void playEffect(Player owner, Player target, StarCard targetStar, GameState state)
    {
        if (owner == null)
            {
                return;
            }

        Card sun = owner.takeFromDropZone("The Sun");

        if (sun == null)
            {
                sun = owner.getDeck().searchAndRemove("The Sun");
            }

        if (sun != null)
            {
                if (sun instanceof StarCard)
                    {
                        StarCard sunStar = (StarCard) sun;
                        StarCard[] field = owner.getField();
                        int emptyPos = -1;

                        for (int i = 0; i < field.length; i++)
                            {
                                if (field[i] == null)
                                    {
                                        emptyPos = i;
                                        break;
                                    }
                            }

                        if (emptyPos < 0)
                            {
                                owner.addToDropZone(sun);
                            }
                        else
                            {
                                sunStar.setOrbitPosition(emptyPos);

                                if (!owner.placeStar(sunStar))
                                    {
                                        owner.addToDropZone(sun);
                                    }
                            }
                    }
                else
                    {
                        owner.addToDropZone(sun);
                    }
            }

        for (StarCard star : owner.getField())
            {
                if (star != null)
                    {
                        star.heal(star.getMaxHp());
                    }
            }
    }

    @Override
    public String getTargetType() {
        return SpecialCard.TARGET_NONE;
    }

    @Override
    public String getPromptMessage() {
        return "Big Bang: นำ The Sun ลงสนามและฟื้นฟู HP ดาวทุกดวง";

    public void playEffectWithCard(Player owner, Player target, Card giveCard, GameState state)
    {
        if (owner == null || target == null || giveCard == null) 
            {
                return;
            }

        if (owner == target) 
            {
                return;
            }

        if (!owner.getHand().contains(giveCard))

            {
                return;
            }

        target.removeCardFromHand(taken);
        owner.removeCardFromHand(giveCard);

        owner.addCardToHand(taken);
        target.addCardToHand(giveCard);
    }

        return SpecialCard.TARGET_ENEMY_PLAYER;
    }
